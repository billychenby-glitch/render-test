const ABC = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';
const json = (data, status = 200) => new Response(JSON.stringify(data), { status, headers: { 'Content-Type': 'application/json; charset=utf-8', 'Cache-Control': 'no-store' } });
const html = (text) => new Response(text, { headers: { 'Content-Type': 'text/html; charset=utf-8' } });

function newCode() {
  const bytes = crypto.getRandomValues(new Uint8Array(12));
  return Array.from(bytes, (b) => ABC[b % 32]).join('');
}

const PLANS = [1, 3, 7, 14, 30, 0]; // 0 = 永久

async function setup(db) {
  await db.batch([
    db.prepare('CREATE TABLE IF NOT EXISTS licenses (code TEXT PRIMARY KEY, days INTEGER NOT NULL, started INTEGER, device TEXT)'),
    db.prepare('CREATE TABLE IF NOT EXISTS blocked (code TEXT PRIMARY KEY)')
  ]);
}

async function openFault(db, body) {
  const code = String(body.code || '').toUpperCase();
  const id = body.id;
  if (!Number.isInteger(id) || id < 0 || id >= DETAILS.length) return json({ error: '内容不存在' }, 400);
  const device = String(body.device || '');
  if (!/^[0-9a-f]{32}$/.test(device)) return json({ error: '无法识别设备，请刷新页面' }, 400);
  const now = Date.now();
  if (await db.prepare('SELECT code FROM blocked WHERE code = ?').bind(code).first()) return json({ error: '兑换码已停用' }, 403);
  // 绑定设备；开始时间只记录一次，解绑后不重新计时
  await db.prepare('UPDATE licenses SET started = COALESCE(started, ?), device = ? WHERE code = ? AND device IS NULL').bind(now, device, code).run();
  const row = await db.prepare('SELECT days, started, device FROM licenses WHERE code = ?').bind(code).first();
  if (!row) return json({ error: '兑换码无效' }, 403);
  if (row.device !== device) return json({ error: '此兑换码已在其他设备使用' }, 403);
  const expires = row.days ? row.started + row.days * 86400000 : 0;
  if (expires && now >= expires) return json({ error: '兑换码已过期' }, 403);
  return json({ html: DETAILS[id], expires });
}

async function makeCodes(env, body) {
  const secret = env.ADMIN_PASSWORD || '';
  if (secret.length < 12) return json({ error: '请先设置至少 12 位的 ADMIN_PASSWORD' }, 500);
  if (body.password !== secret) return json({ error: '密码错误' }, 403);
  if (body.action === 'list') {
    const r = await env.DB.prepare('SELECT l.code, l.days, l.started, l.device IS NOT NULL AS device, b.code IS NOT NULL AS disabled FROM licenses l LEFT JOIN blocked b ON b.code = l.code ORDER BY l.rowid DESC').all();
    return json({ list: r.results });
  }
  if (body.action === 'delete') {
    const list = Array.isArray(body.codes) ? body.codes.map(String) : [];
    // 分批执行，避免单次请求过大
    for (let i = 0; i < list.length; i += 20) {
      await env.DB.batch(list.slice(i, i + 20).flatMap((c) => [
        env.DB.prepare('DELETE FROM licenses WHERE code = ?').bind(c),
        env.DB.prepare('DELETE FROM blocked WHERE code = ?').bind(c)
      ]));
    }
    return json({ ok: true });
  }
  const actions = {
    unbind: 'UPDATE licenses SET device = NULL WHERE code = ?',
    disable: 'INSERT OR IGNORE INTO blocked (code) VALUES (?)',
    enable: 'DELETE FROM blocked WHERE code = ?'
  };
  if (Object.hasOwn(actions, body.action)) {
    await env.DB.prepare(actions[body.action]).bind(String(body.code || '')).run();
    return json({ ok: true });
  }
  if (body.action !== 'create') return json({ error: '请刷新管理页面' }, 400);
  const { count, days } = body;
  if (!(Number.isInteger(count) && count >= 1 && count <= 50 && PLANS.includes(days))) {
    return json({ error: '个数 1~50，天数请从列表选择' }, 400);
  }
  const codes = Array.from({ length: count }, newCode);
  await env.DB.batch(codes.map((c) => env.DB.prepare('INSERT INTO licenses (code, days) VALUES (?, ?)').bind(c, days)));
  return json({ codes });
}

export default {
  async fetch(request, env) {
    const path = new URL(request.url).pathname;
    if (request.method === 'GET' && path === '/') return html(PAGE);
    if (request.method === 'GET' && path === '/admin') return html(ADMIN);
    if (request.method !== 'POST' || (path !== '/api/open' && path !== '/api/admin')) return new Response('Not found', { status: 404 });
    if (!env.DB) return json({ error: '未绑定数据库 DB' }, 500);
    let body;
    try { body = await request.json(); } catch (e) { return json({ error: '请求格式错误' }, 400); }
    if (!body || typeof body !== 'object') return json({ error: '请求格式错误' }, 400);
    await setup(env.DB);
    return path === '/api/open' ? openFault(env.DB, body) : makeCodes(env, body);
  }
};
